## departments

```sql
CREATE TABLE departments (

    CONSTRAINT chk_departments_name CHECK (CHAR_LENGTH(TRIM(department_name)) > 0)
```

## classrooms

```sql
CREATE TABLE classrooms (

    classroom_id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    department_id INT UNSIGNED NOT NULL,
    education_level_id INT UNSIGNED NOT NULL,
    classroom_name VARCHAR(30) NOT NULL,
    CONSTRAINT uq_classrooms_group_name UNIQUE (department_id, education_level_id, classroom_name),
    -- Explicit unique parent key for composite foreign keys in students/book_sets.
    CONSTRAINT uq_classrooms_identity UNIQUE (classroom_id, department_id, education_level_id),
    CONSTRAINT fk_classrooms_department FOREIGN KEY (department_id)
        REFERENCES departments (department_id) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT fk_classrooms_level FOREIGN KEY (education_level_id)
        REFERENCES education_levels (education_level_id) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT chk_classrooms_name CHECK (CHAR_LENGTH(TRIM(classroom_name)) > 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

## students

```sql
CREATE TABLE students (

    student_id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    student_code VARCHAR(20) CHARACTER SET ascii COLLATE ascii_bin NOT NULL,
    password VARCHAR(255) CHARACTER SET ascii COLLATE ascii_bin NOT NULL COMMENT 'Password hash only; generated by backend in Phase 3',
    must_change_password TINYINT(1) NOT NULL DEFAULT 0,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    department_id INT UNSIGNED NOT NULL,
    education_level_id INT UNSIGNED NOT NULL,
    classroom_id INT UNSIGNED NOT NULL,
    phone VARCHAR(20) NULL,
    email VARCHAR(254) NULL,
    CONSTRAINT uq_students_code UNIQUE (student_code),
    KEY idx_students_group (department_id, education_level_id, classroom_id),
    -- Enforces that the stored department and level belong to this classroom.
    CONSTRAINT fk_students_classroom_group FOREIGN KEY (classroom_id, department_id, education_level_id)
        REFERENCES classrooms (classroom_id, department_id, education_level_id)
        ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT chk_students_code CHECK (CHAR_LENGTH(TRIM(student_code)) > 0),
    CONSTRAINT chk_students_password CHECK (CHAR_LENGTH(password) > 0),
    CONSTRAINT chk_students_first_name CHECK (CHAR_LENGTH(TRIM(first_name)) > 0),
    CONSTRAINT chk_students_last_name CHECK (CHAR_LENGTH(TRIM(last_name)) > 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

## book_sets

```sql
CREATE TABLE book_sets (

    book_set_id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    department_id INT UNSIGNED NOT NULL,
    education_level_id INT UNSIGNED NOT NULL,
    classroom_id INT UNSIGNED NOT NULL,
    semester TINYINT UNSIGNED NOT NULL COMMENT '1=first, 2=second, 3=summer',
    academic_year SMALLINT UNSIGNED NOT NULL COMMENT 'Buddhist Era, e.g. 2569',
    CONSTRAINT uq_book_sets_group_term UNIQUE
        (department_id, education_level_id, classroom_id, semester, academic_year),
    CONSTRAINT fk_book_sets_classroom_group FOREIGN KEY (classroom_id, department_id, education_level_id)
        REFERENCES classrooms (classroom_id, department_id, education_level_id)
        ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT chk_book_sets_semester CHECK (semester IN (1, 2, 3)),
    CONSTRAINT chk_book_sets_year CHECK (academic_year BETWEEN 2500 AND 2999)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

## booking_slots

```sql
CREATE TABLE booking_slots (

    slot_id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    booking_date DATE NOT NULL COMMENT 'Gregorian date, e.g. 2026-09-15',
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    max_students SMALLINT UNSIGNED NOT NULL,
    status VARCHAR(10) CHARACTER SET ascii COLLATE ascii_bin NOT NULL DEFAULT 'OPEN',
    CONSTRAINT uq_booking_slots_period UNIQUE (booking_date, start_time, end_time),
    KEY idx_booking_slots_date_status (booking_date, status),
    CONSTRAINT chk_booking_slots_times CHECK
        (start_time >= '00:00:00' AND start_time < end_time AND end_time <= '23:59:59'),
    CONSTRAINT chk_booking_slots_capacity CHECK (max_students > 0),
    CONSTRAINT chk_booking_slots_status CHECK (status IN ('OPEN', 'CLOSED'))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

## bookings

```sql
CREATE TABLE bookings (

    booking_id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    student_id INT UNSIGNED NOT NULL,
    slot_id INT UNSIGNED NOT NULL,
    -- Needed to prevent duplicate bookings for the same student's book set.
    book_set_id INT UNSIGNED NOT NULL,
    queue_number VARCHAR(20) CHARACTER SET ascii COLLATE ascii_bin NOT NULL,
    qr_code VARCHAR(128) CHARACTER SET ascii COLLATE ascii_bin NULL COMMENT 'Opaque token, not an image or personal details',
    booking_status VARCHAR(15) CHARACTER SET ascii COLLATE ascii_bin NOT NULL DEFAULT 'WAITING',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    received_at DATETIME NULL DEFAULT NULL COMMENT 'Actual pickup time in Thailand; NULL until confirmed',
    -- NULL permits cancelled history; all other states retain the one-booking right.
    booking_guard TINYINT GENERATED ALWAYS AS
        (CASE WHEN booking_status = 'CANCELLED' THEN NULL ELSE 1 END) STORED,
    CONSTRAINT uq_bookings_student_set UNIQUE (student_id, book_set_id, booking_guard),
    CONSTRAINT uq_bookings_slot_queue UNIQUE (slot_id, queue_number),
    CONSTRAINT uq_bookings_qr UNIQUE (qr_code),
    KEY idx_bookings_slot_status (slot_id, booking_status),
    KEY idx_bookings_status_created (booking_status, created_at),
    CONSTRAINT fk_bookings_student FOREIGN KEY (student_id)
        REFERENCES students (student_id) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT fk_bookings_slot FOREIGN KEY (slot_id)
        REFERENCES booking_slots (slot_id) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT fk_bookings_set FOREIGN KEY (book_set_id)
        REFERENCES book_sets (book_set_id) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT chk_bookings_status CHECK
        (booking_status IN ('WAITING', 'CHECKED_IN', 'RECEIVED', 'CANCELLED', 'NO_SHOW')),
    CONSTRAINT chk_bookings_queue CHECK (CHAR_LENGTH(TRIM(queue_number)) > 0),
    CONSTRAINT chk_bookings_qr CHECK (qr_code IS NULL OR CHAR_LENGTH(TRIM(qr_code)) > 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```
